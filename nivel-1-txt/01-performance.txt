# --- animações mais rápidas (deixa o sistema "seco") ---
settings put global window_animation_scale 0.5
settings put global transition_animation_scale 0.5
settings put global animator_duration_scale 0.5

# --- força renderização por GPU ---
settings put global force_gpu_rendering 1

# --- remove blur de janelas e transparência (ganha FPS) ---
settings put global disable_window_blurs 1
settings put global accessibility_reduce_transparency 1

# --- mais processos em cache (troca de app mais rápida) ---
settings put global activity_manager_constants max_cached_processes=32

# --- renderizador SkiaGL ---
setprop debug.hwui.renderer skiagl
