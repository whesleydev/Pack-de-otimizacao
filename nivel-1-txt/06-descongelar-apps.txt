# ============================================================
#  NÍVEL 1 · DESCONGELAR APPS
#  Libera de volta os apps congelados pelo 05-congelar-apps.txt
# ============================================================

# --- desliga o congelamento nativo ---
settings put global cached_apps_freezer disabled

# --- descongela um app específico (troque o pacote) ---
pm unsuspend --user 0 com.instagram.android
pm unsuspend --user 0 com.facebook.katana
pm unsuspend --user 0 com.facebook.orca
pm unsuspend --user 0 com.ss.android.ugc.trill
pm unsuspend --user 0 com.google.android.youtube
pm unsuspend --user 0 com.android.chrome
pm unsuspend --user 0 com.spotify.music
pm unsuspend --user 0 com.discord

# --- garante reativação ---
pm enable --user 0 com.instagram.android
pm enable --user 0 com.facebook.katana
pm enable --user 0 com.facebook.orca
pm enable --user 0 com.ss.android.ugc.trill
pm enable --user 0 com.google.android.youtube
pm enable --user 0 com.android.chrome
pm enable --user 0 com.spotify.music
pm enable --user 0 com.discord
