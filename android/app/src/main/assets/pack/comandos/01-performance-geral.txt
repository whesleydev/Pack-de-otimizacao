# --- mais processos em cache (troca de app mais rápida) ---
settings put global activity_manager_constants max_cached_processes=32

# --- renderizador de UI mais rápido ---
setprop debug.hwui.renderer skiagl

# --- desliga o rastreamento de execução (consome CPU) ---
setprop persist.traced.enable 0

# --- NÃO mostrar erro de app em segundo plano ---
settings put global anr_show_background 0

# --- menos diálogos de erro atrapalhando ---
settings put global show_first_crash_dialog 0
