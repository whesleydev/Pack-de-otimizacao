# ============================================================
#  09 · JOGO / GAME MODE
#  Coloca o aparelho em modo jogo e melhora a alocação de recursos.
#  Rode ANTES de abrir o jogo.
# ============================================================

# --- Game Mode em performance ---
cmd game set --mode performance com.dts.freefireth
cmd game set --mode performance com.dts.freefiremax
cmd game set --downscale-factor 1 com.dts.freefireth

# --- força driver de jogo em todos os apps ---
settings put global game_driver_all_apps 1

# --- modo de desempenho fixo (segura o clock alto) ---
cmd power set-fixed-performance-mode-enabled true

# --- fecha apps pesados em segundo plano (respeitando a whitelist) ---
WHITELIST="com.whatsapp com.whatsapp.w4b com.termux com.android.dialer com.android.mms"
for a in com.instagram.android com.facebook.katana com.facebook.orca \
         com.ss.android.ugc.trill com.ss.android.ugc.aweme com.twitter.android \
         com.google.android.youtube com.android.chrome com.spotify.music com.discord; do
    case " $WHITELIST " in *" $a "*) continue ;; esac
    am force-stop "$a"
done
