# ============================================================
#  04 · RAM / MEMÓRIA
#  Controla quantos processos ficam em memória e libera RAM.
#  Mais processos em cache = troca de app rápida.
#  Menos processos = mais RAM livre pro jogo.
#
#  Regra prática: jogo pesado -> perfil GAMING (mais RAM livre).
#  Uso normal/multitarefa -> perfil PADRÃO (troca de app rápida).
#  Escolha UM perfil e cole só ele.
# ============================================================

# ---------- PERFIL PADRÃO (uso normal / multitarefa) ----------
settings put global activity_manager_constants max_cached_processes=32,max_phantom_processes=2147483647

# ---------- PERFIL GAMING (mais RAM pro jogo) ----------
# descomente para usar: menos apps em cache = mais RAM livre
# settings put global activity_manager_constants max_cached_processes=8,max_phantom_processes=1024

# --- encerra todos os apps em segundo plano AGORA ---
am kill-all

# --- limpa o cache de todos os apps ---
pm trim-caches 999999999999

# --- libera memória do sistema (grava buffers no disco) ---
sync
